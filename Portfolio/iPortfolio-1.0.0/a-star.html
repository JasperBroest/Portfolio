Pathfinding!

In this project I made an A-Star pathfinding algorithm in Python. I made this because I wanted to learn more about pathfinding and I chose the A-star algorithm because it is one of the most optimized pathfinding algorithms. This was my first real Python project.

Week 1-2

In the first week I made the a grid with a 2D array. Then I made a breadth first algorithm to find the shortest path from a start point to the end point. First I displayed the path found with directions in the console, and at the end of the week I made the grid visual in the console with the path found.

Here you can see how the main logic works. I check around the current node for any walkable nodes, and put them in a 'to-do' list. I made a node class for ever location on the grid so I can easily get information about the node like if its already walked or if its walkable.

# Make a node class for each position in the grid
class node():
    def __init__(self, x, y, parent=None, action=None):
        self.parent = parent
        self.action = action
        self.x = x
        self.y = y
        self.visited = False
        self.g = 0
        self.h = 0
        self.f = 0

    def calcute_cost(self, goal_node):
        # If has parent then g is parent's g + 1, else g is 0
        self.g = self.parent.g + 1 if self.parent else 0
        # Calculates the distance between the current node and the goal node
        # self.h = abs(goal_node.x - self.x) + abs(goal_node.y - self.y)
        self.h = ((goal_node.x - self.x) ** 2 + (goal_node.y - self.y) ** 2) ** 0.5
        self.f = self.g + self.h
        

    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

def draw_text(text, font, color, x, y):
    img = font.render(text, True, color)
    screen.blit(img, (x, y))

def reset_visual_grid(visual_grid, original_visual_grid):
    for y in range(len(visual_grid)):
        for x in range(len(visual_grid[0])):
            visual_grid[y][x] = original_visual_grid[y][x]

def display_grid(start_x, start_y, goal_x, goal_y, grid, visual_grid):
    x, y = 10, 0
    screen.fill((255, 255, 255))  # Clear the screen
    for item_index, item in enumerate(visual_grid):
        for col_index, col in enumerate(item):
            if item_index == start_y and col_index == start_x:
                draw_text(f"{visual_grid[start_y][start_x]}", text_font, (0, 255, 0), x, y)
            elif item_index == goal_y and col_index == goal_x:
                draw_text(f"{visual_grid[goal_y][goal_x]}", text_font, (255, 0, 0), x, y)
            elif col == '×':
                draw_text(f"{col}", text_font, (0, 0, 255), x, y)
            elif col == '✓':
                draw_text(f"{col}", text_font, (0, 200, 0), x, y)
            elif col == '■':
                draw_text(f"{col}", text_font, (200, 0, 200), x, y)
            elif col == '∎':
                draw_text(f"{col}", text_font, (100, 100, 100), x, y)
            else:
                draw_text(f"{col}", text_font, (0, 0, 0), x, y)
            x += 20
        x = 10
        y += 30
    pygame.display.flip()

def find_path(start_x, start_y, goal_x, goal_y, grid, visual_grid, visualize=True):
    start_node = node(start_x, start_y)
    goal_node = node(goal_x, goal_y)
    to_do = [start_node]
    done = []
    max_width = len(grid[0])
    max_length = len(grid)
    while len(to_do) > 0:
        current = to_do[0]
        for item in to_do:
            item.calcute_cost(goal_node)
            if item.f < current.f:
                current = item
            elif item.f == current.f and item.h < current.h:
                current = item
        to_do.remove(current)
        if current.x == goal_node.x and current.y == goal_node.y:
            to_do.clear()
            path = []
            while current.parent is not None:
                path.append(current)
                current = current.parent
            return path[::-1]
        # Add neighbors
        for dx, dy, action in [(0,1,"down"), (1,0,"right"), (-1,0,"left"), (0,-1,"up")]:
            nx, ny = current.x + dx, current.y + dy
            if 0 <= nx < max_width and 0 <= ny < max_length and grid[ny][nx] == 0:
                neighbor = node(nx, ny, parent=current, action=action)
                if neighbor not in done and neighbor not in to_do:
                    to_do.append(neighbor)
        done.append(current)
        current.visited = True
        if visual_grid[current.y][current.x] not in ('✓', '■'):
            visual_grid[current.y][current.x] = '×'
        if visualize:
            display_grid(start_x, start_y, goal_x, goal_y, grid, visual_grid)
            pygame.time.delay(100)
    return []

screen = pygame.display.set_mode((1024, 768))
text_font = pygame.font.SysFont("Arial", 30)
running = True

map_index = 0

while running and map_index < len(maps):
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

    grid, visual_grid, original_visual_grid, start_x, start_y, goal_x, goal_y = maps[map_index]
    reset_visual_grid(visual_grid, original_visual_grid)
    display_grid(start_x, start_y, goal_x, goal_y, grid, visual_grid)
    pygame.time.delay(500)

    # Find path and visualize search
    path = find_path(start_x, start_y, goal_x, goal_y, grid, visual_grid, visualize=True)
    # Animate the path with checkmarks
    for item in path:
        visual_grid[item.y][item.x] = '✓'
        display_grid(start_x, start_y, goal_x, goal_y, grid, visual_grid)
        pygame.time.delay(80)
    pygame.time.delay(800)
    map_index += 1

pygame.quit()
#end region pygame
